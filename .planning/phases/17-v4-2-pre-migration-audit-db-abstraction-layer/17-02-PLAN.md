          class Transaction:
              """Light facade for use inside `async with db.transaction()`. In Phase 21 becomes asyncpg connection wrapper."""
              def __init__(self, conn: "aiosqlite.Connection") -> None:
                  self._conn = conn

              async def execute(self, sql: str, params: tuple[Any, ...] = ()) -> None:
                  try:
                      await self._conn.execute(sql, params)
                  except aiosqlite.Error as e:
                      raise DatabaseError(str(e)) from e

              async def fetch_one(self, sql: str, params: tuple[Any, ...] = ()) -> dict | None:
                  try:
                      cur = await self._conn.execute(sql, params)
                      row = await cur.fetchone()
                      await cur.close()
                      return dict(row) if row else None
                  except aiosqlite.Error as e:
                      raise DatabaseError(str(e)) from e

              async def fetch_all(self, sql: str, params: tuple[Any, ...] = ()) -> list[dict]:
                  try:
                      cur = await self._conn.execute(sql, params)
                      rows = await cur.fetchall()
                      await cur.close()
                      return [dict(r) for r in rows]
                  except aiosqlite.Error as e:
                      raise DatabaseError(str(e)) from e
          ```

       g) Add `transaction` method on DatabaseManager:

          @asynccontextmanager
          async def transaction(self) -> AsyncIterator[Transaction]:
              if self._conn is None:
                  raise DatabaseError("Database not connected")
              try:
                  await asyncio.wait_for(self._tx_lock.acquire(), timeout=10.0)
              except asyncio.TimeoutError as e:
                  logger.error("transaction(): _tx_lock timeout >10s — possible deadlock")
                  raise DatabaseError("transaction lock timeout") from e
              try:
                  await self._conn.execute("BEGIN IMMEDIATE")
                  try:
                      yield Transaction(self._conn)
                      await self._conn.commit()
                  except Exception:
                      await self._conn.rollback()
                      raise
              except aiosqlite.Error as e:
                  raise DatabaseError(str(e)) from e
              finally:
                  self._tx_lock.release()
          ```
